package k8sengine

import corev1 "k8s.io/api/core/v1"

// Normalize installer-managed Ray templates before native admission validates
// every template and descendant. Admission remains the enforcement boundary.
func secureRuntimeTemplate(p *corev1.PodTemplateSpec) {
	p.Spec.ServiceAccountName = "michelangelo-lifecycle-runner"
	p.Spec.AutomountServiceAccountToken = boolPointer(true)
	p.Spec.ImagePullSecrets = []corev1.LocalObjectReference{{Name: "ma-lifecycle-runtime-registry"}}
	p.Spec.SecurityContext = &corev1.PodSecurityContext{RunAsNonRoot: boolPointer(true), RunAsUser: intPointer(65534), FSGroup: intPointer(65534), SeccompProfile: &corev1.SeccompProfile{Type: corev1.SeccompProfileTypeRuntimeDefault}}
	for i := range p.Spec.Volumes {
		if p.Spec.Volumes[i].Name == "ray" && p.Spec.Volumes[i].HostPath != nil {
			p.Spec.Volumes[i].VolumeSource = corev1.VolumeSource{EmptyDir: &corev1.EmptyDirVolumeSource{}}
		}
	}
	for i := range p.Spec.Containers {
		c := &p.Spec.Containers[i]
		c.SecurityContext = &corev1.SecurityContext{AllowPrivilegeEscalation: boolPointer(false), RunAsNonRoot: boolPointer(true), Capabilities: &corev1.Capabilities{Drop: []corev1.Capability{"ALL"}}}
		c.EnvFrom = []corev1.EnvFromSource{{ConfigMapRef: &corev1.ConfigMapEnvSource{LocalObjectReference: corev1.LocalObjectReference{Name: "ma-lifecycle-runtime-artifacts"}}}, {SecretRef: &corev1.SecretEnvSource{LocalObjectReference: corev1.LocalObjectReference{Name: "ma-lifecycle-runtime-artifacts"}}}}
	}
}
func boolPointer(v bool) *bool  { return &v }
func intPointer(v int64) *int64 { return &v }
